<head>
  <title>Twister Letters</title>
  <style>
    /* HTML + CSS (from your PDF "Other requirements") */
    body {
      background-color: white; /* White background */
      margin: 0;
      overflow: hidden; /* Prevent scrollbars */
    }
    .letter {
      position: absolute; /* Allows free movement */
      font-size: 32px;
      font-weight: bold;
      color: black; /* Black text */
    }
  </style>

<body>
  <!-- Instead of one block of text, we split your name into letters -->
  <span class="letter">K</span>
  <span class="letter">a</span>
  <span class="letter">n</span>
  <span class="letter">e</span>

  <script>
    /* JavaScript (also listed in your PDF) runs in the browser, 
       similar to how PHP runs on the server. */

    const letters = document.querySelectorAll(".letter");
    let angle = 0;
    let y = window.innerHeight;

    function animate() {
      // Move upwards
      y -= 2;
      if (y < -50) {
        y = window.innerHeight; // Reset to bottom
      }

      // Each letter gets its own offset, so they swirl independently
      letters.forEach((letter, index) => {
        const offsetAngle = angle + index * 0.5; // stagger each letter
        const xOffset = Math.sin(offsetAngle) * 100;

        letter.style.top = y + "px";
        letter.style.left = (window.innerWidth / 2 + xOffset + index * 20) + "px";
      });

      // Looping animation (like PHP loops in your PDF examples)
      angle += 0.05;
      requestAnimationFrame(animate);
    }

    animate(); // Start animation
  </script>
</body>
