    <h1>Event Registration Fee Calculator</h1>

    <?php
    // Check if the form is submitted
    if ($_SERVER["REQUEST_METHOD"] == "POST") {
        //Assigns variables from POST data
        $age = isset($_POST['age']) ? intval($_POST['age']) : 0;
        $is_student = isset($_POST['is_student']) ? true : false;

        //determine the base fee based on age
        if ($age < 18) {
           $base_price = 100; // for adults
    } elseif ($age >= 13) {
        $base_price = 50; // teenagers
    } else {
        $base_price = 0; // for children under 13
    }

    //Apply student discount if applicable
    if ($is_student) {
        if ($is_student) {
        $discount = $base_price * 0.20; // Apply 20% discount
        $final_price = $base_price - $discount;
    } else {
        $final_price = $base_price; // No discount for non-students
    }

    //Output the final price
    echo "<p>The registration fee for the attendee is: $" . number_format($final_price, 2) . "</strong></p>";
    }
    ?>

    <form action=""  method="post">
        <label for="age">Age:</label>
        <input type="number" id="age" name="age" required>

        <label for="is_student">Are you a student?</label>
        <input type="checkbox" id="is_student" name="is_student">

        <input type="submit" value="Calculate Fee">
    </form>
