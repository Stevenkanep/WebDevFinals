?>
<!DOCTYPE html>
<html>
<head>
  <title>4.0 Tasker</title>
  <style>
    /* CSS styling (from your module: HTML + CSS as "Other requirements") */
    body {
      background-color: white; 
      margin: 0;
      overflow: hidden; 
    }
    .letter {
      position: absolute; 
      font-size: 32px;
      font-weight: bold;
      color: black; 
    }
  </style>
</head>
<body>
<?php
// Instead of writing <span> directly, we can echo them with PHP.
// This shows how PHP outputs HTML, just like the "echo 'Welcome to PHP';" example in your PDF.
$name = "STEVENKANE PABELONA"; 
$letters = str_split($name);
foreach ($letters as $ch) {
    if ($ch !== " ") { // skip spaces
        echo "<span class='letter'>$ch</span>";
    }
}
?>

<script>
  // JavaScript handles the animation (like client-side scripting in your PDF).
  const letters = document.querySelectorAll(".letter");
  let angle = 0;
  let y = window.innerHeight;

  function animate() {
    y -= 2;
    if (y < -50) {
      y = window.innerHeight; 
    }

    // Each letter gets its own staggered offset, creating a trailing twister effect
    letters.forEach((letter, index) => {
      const offsetAngle = angle + index * 0.3; // smaller step for smoother trail
      const xOffset = Math.sin(offsetAngle) * 120;

      letter.style.top = (y - index * 15) + "px"; // stagger vertically
      letter.style.left = (window.innerWidth / 2 + xOffset) + "px";
    });

    angle += 0.05;
    requestAnimationFrame(animate);
  }

  animate();
</script>
</body>
</html>
