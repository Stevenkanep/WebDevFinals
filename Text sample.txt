<?php
$name = "STEVENKANE PABELONA"; 
$letters = str_split($name);
foreach ($letters as $ch) {
    if ($ch !== " ") { 
        echo "<span class='letter'>$ch</span>";
    }
}
?>

<script>
  const letters = document.querySelectorAll(".letter");
  let angle = 0;
  let y = window.innerHeight;

  function animate() {
    y -= 2;
    if (y < -50) {
      y = window.innerHeight; 
    }

    letters.forEach((letter, index) => {
      // Spiral effect: use both sine and cosine for circular motion
      const offsetAngle = angle + index * 0.5; 
      const radius = 100 + index * 5; // radius grows with index for spiral
      const xOffset = Math.cos(offsetAngle) * radius;
      const yOffset = Math.sin(offsetAngle) * radius;

      // Position letters in a spiral while drifting upward
      letter.style.top = (y + yOffset - index * 10) + "px"; 
      letter.style.left = (window.innerWidth / 2 + xOffset) + "px";
    });

    angle += 0.05;
    requestAnimationFrame(animate);
  }

  animate();
</script>
